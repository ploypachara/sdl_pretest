function getScores() {

    const result = {};

    categories.forEach(cat => {

        const vals = [];

        cat.items.forEach((_, i) => {

            const radios = document.querySelectorAll(`input[name="${cat.id}_q${i}"]`);

            let selected = null;

            radios.forEach(r => { if (r.checked) selected = parseInt(r.value); });

            vals.push(selected);

        });

        result[cat.id] = vals;

    });

    return result;

}


function validateAll() {

    const ids = ['prefix','fullname','email','organization','role'];

    for (let id of ids) {

        if (!document.getElementById(id).value.trim()) {

            showStatus('result-status-msg', 'กรุณากรอกข้อมูลส่วนตัวให้ครบถ้วน', 'error');

            return false;

        }

    }

    if (!document.getElementById('occupation').value) {

        showStatus('result-status-msg', 'กรุณาเลือกอาชีพ/สายงาน', 'error');

        return false;

    }

    const scores = getScores();

    for (let catId in scores) {

        for (let v of scores[catId]) {

            if (v === null) {

                showStatus('result-status-msg', `กรุณาตอบคำถามทุกข้อในหมวด ${catId} ให้ครบถ้วน`, 'error');

                return false;

            }

        }

    }

    return true;

}

function calculateResults() {

    document.getElementById('result-status-msg').className = 'status-msg';

    document.getElementById('result-status-msg').style.display = 'none';

    if (!validateAll()) return;


    const scores = getScores();

    const prefix = document.getElementById('prefix').value.trim();

    const fullname = document.getElementById('fullname').value.trim();

    document.getElementById('profile-summary').style.display = 'block';

    document.getElementById('profile-summary').innerHTML = `👤 <strong>${prefix}${fullname}</strong> | <strong>อาชีพ:</strong> ${document.getElementById('occupation').value}`;


    let html = '';

    categories.forEach(cat => {

        const vals = scores[cat.id];

        const avg = vals.reduce((a,b) => a+b, 0) / vals.length;

        const avgFixed = avg.toFixed(2);

        const interp = interpretDesc(cat.id, avg);

        let levelClass = avg >= 4.0 ? 'high' : (avg <= 2.4 ? 'low' : 'medium');

        html += `<div class="result-card ${levelClass}" style="border-left-color:${cat.color};">

            <div style="display:flex; justify-content:space-between; align-items:center;">

                <div style="font-weight:700; font-size:0.95em;">${cat.name}</div>

                <div><span class="score">${avgFixed}</span> <span class="score-label">/ 5</span></div>

            </div>

            <div class="desc">${interp.desc}</div>

            <div class="improve">💡 ${interp.improve}</div>

        </div>`;

    });

    document.getElementById('result-grid').innerHTML = html;

    switchTab('result');

}


function saveResult() {

    const btn = document.getElementById('btn-save');

    btn.disabled = true;

    btn.innerHTML = '<span class="loading"></span> กำลังบันทึก...';

    

    const scores = getScores();

    const prefix = document.getElementById('prefix').value.trim();

    const fullname = document.getElementById('fullname').value.trim();

    const email = document.getElementById('email').value.trim();

    const org = document.getElementById('organization').value.trim();

    const occ = document.getElementById('occupation').value;

    const role = document.getElementById('role').value.trim();

    

    if (!prefix||!fullname||!email||!org||!occ||!role) {

        showStatus('result-status-msg','กรุณากรอกข้อมูลส่วนตัวให้ครบถ้วน','error');

        btn.disabled = false; btn.innerHTML = '💾 บันทึกผลลงฐานข้อมูล';

        return;

    }

    for (let catId in scores) {

        for (let v of scores[catId]) {

            if (v === null) { 

                showStatus('result-status-msg','กรุณาตอบทุกข้อก่อนบันทึก','error');

                btn.disabled = false; btn.innerHTML = '💾 บันทึกผลลงฐานข้อมูล';

                return; 

            }

        }

    }

    

    const now = new Date();

    const thaiTime = now.toLocaleString('th-TH',{timeZone:'Asia/Bangkok',year:'numeric',month:'2-digit',day:'2-digit',hour:'2-digit',minute:'2-digit',second:'2-digit'});

    

    const payload = {

        timestamp: thaiTime,

        prefix, fullname, email, org, occ, role,

        scores: JSON.parse(JSON.stringify(scores))

    };

    

    fetch(APPS_SCRIPT_URL, {

        method: 'POST',

        headers: { 'Content-Type': 'application/json' },

        body: JSON.stringify(payload)

    })

    .then(() => {

        showStatus('result-status-msg',`✅ บันทึกข้อมูลของ ${prefix}${fullname} สำเร็จ! (${thaiTime})`,'success');

        btn.disabled = false; btn.innerHTML = '💾 บันทึกผลลงฐานข้อมูล';

    })

    .catch((err) => {

        showStatus('result-status-msg',`❌ เกิดข้อผิดพลาด: ${err.message}`,'error');

        btn.disabled = false; btn.innerHTML = '💾 บันทึกผลลงฐานข้อมูล';

    });

}

function fetchAndExport() {

    fetch(APPS_SCRIPT_URL + '?t=' + Date.now())

    .then(res => res.json())

    .then(data => {

        if (data.success && data.csv) {

            const blob = new Blob([data.csv], {type: 'text/csv;charset=utf-8;'});

            const link = document.createElement('a');

            link.href = URL.createObjectURL(blob);

            link.download = `SDL_Pretest_Data_${new Date().toISOString().slice(0,10)}.csv`;

            link.click();

            URL.revokeObjectURL(link.href);

            showStatus('admin-status-msg',`✅ ดาวน์โหลด CSV สำเร็จ — ${data.data.length} รายการ`,'success');

        } else {

            showStatus('admin-status-msg', '❌ ไม่มีข้อมูลให้ดาวน์โหลด','error');

        }

    })

    .catch(err => {

        showStatus('admin-status-msg', '❌ เกิดข้อผิดพลาด: '+err.message,'error');

    });

}

function switchTab(tab) {

        document.getElementById('result-status-msg').className = 'status-msg';

        document.getElementById('result-status-msg').style.display = 'none';

    }

}
